## 1. PRE-REQUISITE: Implement `encrypt` in `paperpilot-wasm/src/operations.rs`

In `paperpilot-wasm/src/operations.rs`, `encrypt` currently returns `Err(PdfError::UnsupportedOperation)`.
You MUST implement real pure-Rust in-memory encryption using `lopdf` (matching `paperpilot-pdf/src/operations/encrypt.rs`):

```rust
use lopdf::{EncryptionState, EncryptionVersion, Permissions};
use std::convert::TryFrom;

pub fn encrypt(input_bytes: &[u8], password: &str) -> OperationResult<Vec<u8>> {
    let mut doc = Document::load_mem(input_bytes)
        .map_err(|e| PdfError::ParseError(format!("Failed to parse document: {e}")))?;

    // Dynamically inject a placeholder ID array if missing
    if doc.trailer.get(b"ID").is_err() {
        let id_str = Object::String(
            b"default_id_placeholder".to_vec(),
            lopdf::StringFormat::Literal,
        );
        doc.trailer.set("ID", Object::Array(vec![id_str.clone(), id_str]));
    }

    let version = EncryptionVersion::V2 {
        document: &doc,
        owner_password: password,
        user_password: password,
        key_length: 128,
        permissions: Permissions::all(),
    };

    let state = EncryptionState::try_from(version)
        .map_err(|e| PdfError::Other(format!("Encryption failed: {e}")))?;

    doc.encrypt(&state)
        .map_err(|e| PdfError::Other(format!("Encryption failed: {e}")))?;

    let mut buffer = Vec::new();
    doc.save_to(&mut buffer)
        .map_err(|e| PdfError::IoError(std::io::Error::new(std::io::ErrorKind::Other, e.to_string())))?;

    Ok(buffer)
}
```

## 3. Implementation Details: `apps/web/tests/e2e_wasm_12_tools.spec.ts`

### Test Architecture
1. **Fixture Files**:
   Read test fixtures from `../../tests/e2e_fixtures/` using `fs.readFileSync`:
   - `single_page.pdf` (539 bytes)
   - `multi_page.pdf` (1,409 bytes, 2 pages)

2. **Real Web Worker Execution**:
   In each test step, navigate to `page.goto('/')`, and execute the operation directly through `wasmPdfClient` inside the browser context:
   ```typescript
   const result = await page.evaluate(async ({ opName, payload }) => {
     const { wasmPdfClient } = await import('/src/lib/wasm/index.ts');
     let res: Uint8Array | Uint8Array[];
     if (opName === 'merge') {
       res = await wasmPdfClient.merge(payload.files.map((b: string) => Uint8Array.from(atob(b), c => c.charCodeAt(0))));
     } else if (opName === 'split') {
       res = await wasmPdfClient.split(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.ranges);
     } else if (opName === 'rotate') {
       res = await wasmPdfClient.rotate(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.angle, payload.pages);
     } else if (opName === 'compress') {
       res = await wasmPdfClient.compress(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)));
     } else if (opName === 'encrypt') {
       res = await wasmPdfClient.encrypt(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.password);
     } else if (opName === 'watermark') {
       res = await wasmPdfClient.watermark(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.text);
     } else if (opName === 'delete_pages') {
       res = await wasmPdfClient.delete_pages(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.pages);
     } else if (opName === 'extract_pages') {
       res = await wasmPdfClient.extract_pages(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.pages);
     } else if (opName === 'reorder_pages') {
       res = await wasmPdfClient.reorder_pages(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.new_order);
     } else if (opName === 'crop') {
       res = await wasmPdfClient.crop(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.left, payload.bottom, payload.right, payload.top);
     } else if (opName === 'flatten') {
       res = await wasmPdfClient.flatten(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)));
     } else if (opName === 'set_metadata') {
       res = await wasmPdfClient.set_metadata(Uint8Array.from(atob(payload.file), c => c.charCodeAt(0)), payload.title, payload.author, payload.subject, payload.keywords);
     } else {
       throw new Error(`Unsupported tool: ${opName}`);
     }
     
     const bytes = Array.isArray(res) ? res[0] : res;
     const isPdf = bytes[0] === 0x25 && bytes[1] === 0x50 && bytes[2] === 0x44 && bytes[3] === 0x46; // %PDF
     return { length: bytes.length, isPdf };
   }, { opName, payload });
   ```
