## Breakpoints to Support

| Name | Width | Behaviour |
|---|---|---|
| Desktop | ≥ 900px | Current layout — sidebar always visible on left, ops panel on right |
| Tablet | 600px – 899px | Sidebar collapses to icon-only strip (48px wide), ops panel below content |
| Mobile | < 600px | Sidebar is hidden, replaced by a bottom nav bar; ops panel is a slide-up drawer |

### 2. `apps/desktop/src/lib/components/layout/Sidebar.svelte`

The sidebar needs to track whether it's in collapsed mode for accessibility (aria attributes). Add a reactive `isCollapsed` boolean using `$state` that reads window width via `window.innerWidth`. Use Svelte's `onMount` to set it and add a `resize` event listener.

```svelte
<script lang="ts">
  import { onMount } from 'svelte';
  // ... existing imports

  let windowWidth = $state(typeof window !== 'undefined' ? window.innerWidth : 1200);
  let isCollapsed = $derived(windowWidth < 900);

  onMount(() => {
    function handleResize() {
      windowWidth = window.innerWidth;
    }
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  });
</script>

<aside class="sidebar" aria-label="Main navigation" aria-expanded={!isCollapsed}>
  <div class="sidebar-header">
    <h1 class="brand-title">PaperPilot</h1>
  </div>

  <nav class="sidebar-nav" aria-label="Navigation">
    {#each navItems as item (item.id)}
      <button
        class="nav-item"
        class:active={appState.activeTab === item.id}
        onclick={() => appState.setActiveTab(item.id)}
        title={isCollapsed ? item.label : undefined}
        id={`nav-${item.id}`}
        aria-label={item.label}
        aria-current={appState.activeTab === item.id ? 'page' : undefined}
      >
        <span class="nav-icon" aria-hidden="true">{item.icon}</span>
        <span class="nav-label">{item.label}</span>
      </button>
    {/each}
  </nav>

  <!-- Keep jobs and footer but they're hidden via CSS at tablet/mobile -->
  ...existing jobs and footer sections...
</aside>
```

### 3. New component: `apps/desktop/src/lib/components/layout/BottomNav.svelte`

This component renders a bottom navigation bar for mobile (< 600px), replacing the hidden sidebar. It shows all nav items as centered icon+label buttons.

```svelte
<script lang="ts">
  import { appState } from '$lib/state/app.svelte';
  import { onMount } from 'svelte';

  let windowWidth = $state(typeof window !== 'undefined' ? window.innerWidth : 1200);
  let isMobile = $derived(windowWidth < 600);

  onMount(() => {
    function handleResize() { windowWidth = window.innerWidth; }
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  });

  const navItems = [
    { id: 'home', label: 'Home', icon: '🏠' },
    { id: 'documents', label: 'Documents', icon: '📄' },
    { id: 'settings', label: 'Settings', icon: '⚙️' }
  ];
</script>

{#if isMobile}
  <nav class="bottom-nav" id="mobile-bottom-nav" aria-label="Mobile navigation">
    {#each navItems as item (item.id)}
      <button
        class="bottom-nav-item"
        class:active={appState.activeTab === item.id}
        id={`mobile-nav-${item.id}`}
        onclick={() => appState.setActiveTab(item.id)}
        aria-label={item.label}
        aria-current={appState.activeTab === item.id ? 'page' : undefined}
      >
        <span class="bottom-nav-icon" aria-hidden="true">{item.icon}</span>
        <span class="bottom-nav-label">{item.label}</span>
      </button>
    {/each}
  </nav>
{/if}

<style>
```
