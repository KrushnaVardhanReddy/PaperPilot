### 2. Crate Configuration (`paperpilot-wasm/Cargo.toml`)

### 3. In-Memory API Surface (`paperpilot-wasm/src/lib.rs`)
Implement the `WasmPdfEngine` struct with `wasm-bindgen`:

```rust
use wasm_bindgen::prelude::*;

#[wasm_bindgen]
pub struct WasmPdfEngine;

#[wasm_bindgen]
impl WasmPdfEngine {
    /// Merge multiple PDF buffers into a single PDF
    #[wasm_bindgen]
    pub fn merge(buffers: js_sys::Array) -> Result<js_sys::Uint8Array, JsValue>;

    /// Rotate specific pages or 'all' pages by angle (90, 180, 270)
    #[wasm_bindgen]
    pub fn rotate(input_bytes: &[u8], angle: u16, pages: &str) -> Result<js_sys::Uint8Array, JsValue>;

    /// Split PDF by page ranges (e.g. "1-2, 3-4") returning an array of Uint8Array chunks
    #[wasm_bindgen]
    pub fn split(input_bytes: &[u8], ranges: &str) -> Result<js_sys::Array, JsValue>;

    /// Compress PDF by removing unneeded objects and deflating streams
    #[wasm_bindgen]
    pub fn compress(input_bytes: &[u8]) -> Result<js_sys::Uint8Array, JsValue>;

    /// Encrypt PDF with password
    #[wasm_bindgen]
    pub fn encrypt(input_bytes: &[u8], password: &str) -> Result<js_sys::Uint8Array, JsValue>;

    /// Add text watermark
    #[wasm_bindgen]
    pub fn watermark(input_bytes: &[u8], text: &str) -> Result<js_sys::Uint8Array, JsValue>;
}
```

### 4. Zero Filesystem Footprint
- Never call `std::fs::*`.
- Load and parse directly using `lopdf::Document::load_from(input_bytes)`.
- Save output into a `Vec<u8>` cursor (`doc.save_to(&mut buffer)`), converting back to `js_sys::Uint8Array`.
