# Task 103A: Embedded libSQL Core Storage Engine & Schema Migrations

## Context & Objectives
You are creating a new isolated workspace crate `paperpilot-storage` that embeds **Turso / libSQL** (`libsql = "0.10.0"`) to provide air-gapped local persistence for asynchronous jobs, outbound webhooks, and audit compliance logging.
This crate does NOT touch any PDF processing logic and will be completely self-contained.

## Implementation Details

### 2. Error Definitions (`paperpilot-storage/src/error.rs`)
```rust
use thiserror::Error;

#[derive(Error, Debug)]
pub enum StorageError {
    #[error("Database error: {0}")]
    Db(#[from] libsql::Error),
    #[error("Serialization error: {0}")]
    Serialization(#[from] serde_json::Error),
    #[error("Job not found: {0}")]
    NotFound(String),
}
```

### 3. Models (`paperpilot-storage/src/models.rs`)
Define structs with `Serialize, Deserialize, Clone, Debug`:
- `JobStatus`: enum `Pending`, `Running`, `Completed`, `Failed`.
- `JobRecord`: `id: String`, `operation: String`, `status: JobStatus`, `input_path: Option<String>`, `output_path: Option<String>`, `parameters_json: String`, `error_message: Option<String>`, `progress_percentage: u8`, `callback_url: Option<String>`, `created_at: String`, `updated_at: String`.
- `NewJob`: `operation: String`, `input_path: Option<String>`, `output_path: Option<String>`, `parameters_json: Option<String>`, `callback_url: Option<String>`.
- `AuditEvent`: `id: String`, `event_type: String`, `tool_id: String`, `actor: String`, `payload_hash: String`, `details_json: String`, `timestamp: String`.

### 4. Storage Engine Core (`paperpilot-storage/src/lib.rs`)
```rust
pub mod error;
pub mod models;

use error::StorageError;
use models::*;
use std::path::Path;

pub struct StorageEngine {
    db: libsql::Database,
    conn: libsql::Connection,
}

impl StorageEngine {
    pub async fn open_local(path: &Path) -> Result<Self, StorageError> {
        let path_str = path.to_string_lossy().to_string();
        let db = libsql::Builder::new_local(path_str).build().await?;
        let conn = db.connect()?;
        let engine = Self { db, conn };
        engine.run_migrations().await?;
        Ok(engine)
    }

    pub async fn open_in_memory() -> Result<Self, StorageError> {
        let db = libsql::Builder::new_local(":memory:").build().await?;
        let conn = db.connect()?;
        let engine = Self { db, conn };
        engine.run_migrations().await?;
        Ok(engine)
    }

    pub async fn run_migrations(&self) -> Result<(), StorageError> {
        self.conn.execute_batch(r#"
            CREATE TABLE IF NOT EXISTS jobs (
                id TEXT PRIMARY KEY,
                operation TEXT NOT NULL,
                status TEXT NOT NULL,
                input_path TEXT,
                output_path TEXT,
                parameters_json TEXT NOT NULL DEFAULT '{}',
                error_message TEXT,
                progress_percentage INTEGER NOT NULL DEFAULT 0,
                callback_url TEXT,
                created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
                updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
            );

            CREATE TABLE IF NOT EXISTS webhooks (
                id TEXT PRIMARY KEY,
                name TEXT NOT NULL,
                target_url TEXT NOT NULL,
                secret_token TEXT,
                events_json TEXT NOT NULL,
                is_active BOOLEAN NOT NULL DEFAULT 1,
                retry_count INTEGER NOT NULL DEFAULT 3,
                created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
            );

            CREATE TABLE IF NOT EXISTS document_chunks (
                id TEXT PRIMARY KEY,
                document_hash TEXT NOT NULL,
                document_name TEXT NOT NULL,
                page_number INTEGER NOT NULL,
                chunk_index INTEGER NOT NULL,
                content TEXT NOT NULL,
                embedding_json TEXT NOT NULL,
                created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
            );

            CREATE TABLE IF NOT EXISTS audit_logs (
                id TEXT PRIMARY KEY,
                event_type TEXT NOT NULL,
                tool_id TEXT NOT NULL,
                actor TEXT NOT NULL DEFAULT 'local_user',
                payload_hash TEXT NOT NULL,
                details_json TEXT NOT NULL DEFAULT '{}',
                timestamp DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP
            );
        "#).await?;
        Ok(())
    }

    pub async fn create_job(&self, job: &NewJob) -> Result<JobRecord, StorageError> {
        let id = uuid::Uuid::new_v4().to_string();
        let now = chrono::Utc::now().to_rfc3339();
        let params = job.parameters_json.clone().unwrap_or_else(|| "{}".to_string());

        self.conn.execute(
            "INSERT INTO jobs (id, operation, status, input_path, output_path, parameters_json, progress_percentage, callback_url, created_at, updated_at)
             VALUES (?1, ?2, 'pending', ?3, ?4, ?5, 0, ?6, ?7, ?7)",
            libsql::params![
                id.clone(),
                job.operation.clone(),
                job.input_path.clone(),
                job.output_path.clone(),
                params.clone(),
                job.callback_url.clone(),
                now.clone(),
            ]
        ).await?;

        Ok(JobRecord {
            id,
            operation: job.operation.clone(),
            status: JobStatus::Pending,
            input_path: job.input_path.clone(),
            output_path: job.output_path.clone(),
            parameters_json: params,
            error_message: None,
            progress_percentage: 0,
            callback_url: job.callback_url.clone(),
            created_at: now.clone(),
            updated_at: now,
        })
    }

    pub async fn update_job_status(
        &self,
        id: &str,
        status: JobStatus,
        progress: u8,
        output: Option<&str>,
        error: Option<&str>,
    ) -> Result<(), StorageError> {
        let now = chrono::Utc::now().to_rfc3339();
        let status_str = match status {
            JobStatus::Pending => "pending",
            JobStatus::Running => "running",
            JobStatus::Completed => "completed",
            JobStatus::Failed => "failed",
        };

        self.conn.execute(
            "UPDATE jobs SET status = ?1, progress_percentage = ?2, output_path = COALESCE(?3, output_path), error_message = ?4, updated_at = ?5 WHERE id = ?6",
            libsql::params![status_str, progress as i64, output.map(|s| s.to_string()), error.map(|s| s.to_string()), now, id.to_string()]
        ).await?;

        Ok(())
    }

    pub async fn get_job(&self, id: &str) -> Result<Option<JobRecord>, StorageError> {
        let mut rows = self.conn.query(
            "SELECT id, operation, status, input_path, output_path, parameters_json, error_message, progress_percentage, callback_url, created_at, updated_at FROM jobs WHERE id = ?1",
            libsql::params![id.to_string()]
        ).await?;

        if let Some(row) = rows.next().await? {
            let status_str: String = row.get(2)?;
            let status = match status_str.as_str() {
                "running" => JobStatus::Running,
                "completed" => JobStatus::Completed,
                "failed" => JobStatus::Failed,
                _ => JobStatus::Pending,
            };

            let prog: i64 = row.get(7)?;
            Ok(Some(JobRecord {
                id: row.get(0)?,
                operation: row.get(1)?,
                status,
                input_path: row.get(3).ok(),
                output_path: row.get(4).ok(),
                parameters_json: row.get(5)?,
                error_message: row.get(6).ok(),
                progress_percentage: prog as u8,
                callback_url: row.get(8).ok(),
                created_at: row.get(9)?,
                updated_at: row.get(10)?,
            }))
        } else {
            Ok(None)
        }
    }

    pub async fn record_audit_event(&self, event: &AuditEvent) -> Result<(), StorageError> {
        self.conn.execute(
            "INSERT INTO audit_logs (id, event_type, tool_id, actor, payload_hash, details_json, timestamp)
             VALUES (?1, ?2, ?3, ?4, ?5, ?6, ?7)",
            libsql::params![
                event.id.clone(),
                event.event_type.clone(),
                event.tool_id.clone(),
                event.actor.clone(),
                event.payload_hash.clone(),
                event.details_json.clone(),
                event.timestamp.clone(),
            ]
        ).await?;
        Ok(())
    }
}
```
