## CONTEXT

### File to Modify
apps/desktop/src/lib/components/layout/Sidebar.svelte

### Current Behavior
The sidebar is currently 260px wide (fixed). It only "collapses" automatically
based on window width when windowWidth < 900 via:
  `let isCollapsed = $derived(windowWidth < 900);`
There is NO manual toggle for the user.

### Goal
Add a manual collapse toggle button that:
1. Collapses the sidebar to a 48px icon-only "rail" (showing just nav icons)
2. Has a smooth CSS width transition
3. Persists the user's choice in localStorage
4. Still auto-collapses at small window widths (preserve existing behavior)

## CHANGE 1: Update Script Logic

### Current script block (lines 1-26):
```typescript
<script lang="ts">
  import { onMount } from 'svelte';
  import { appState } from '$lib/state/app.svelte';
  import { jobsState } from '$lib/state/jobs.svelte';
  import ProgressBar from '$lib/components/ui/ProgressBar.svelte';
  import CancelButton from '$lib/components/ui/CancelButton.svelte';

  let windowWidth = $state(typeof window !== 'undefined' ? window.innerWidth : 1200);
  let isCollapsed = $derived(windowWidth < 900);

  onMount(() => {
    function handleResize() {
      windowWidth = window.innerWidth;
    }
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  });
  ...
```

### Replace ONLY the state variables and onMount (keep all imports and navItems):
```typescript
  let windowWidth = $state(typeof window !== 'undefined' ? window.innerWidth : 1200);
  let manualCollapsed = $state(false);
  let isCollapsed = $derived(manualCollapsed || windowWidth < 900);

  onMount(() => {
    // Restore persisted collapse state
    const stored = localStorage.getItem('sidebar-collapsed');
    if (stored === 'true') manualCollapsed = true;

    function handleResize() {
      windowWidth = window.innerWidth;
    }
    window.addEventListener('resize', handleResize);
    return () => window.removeEventListener('resize', handleResize);
  });

  $effect(() => {
    localStorage.setItem('sidebar-collapsed', String(manualCollapsed));
  });
```

  id="sidebar-collapse-toggle"
