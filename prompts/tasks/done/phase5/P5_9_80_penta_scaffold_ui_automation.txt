### Step 1: Update Scaffolding Generator (`tools/penta-scaffold/src/generator.rs`)

Implement `append_desktop_args_builder`:
```rust
fn append_desktop_args_builder(args: &ScaffoldArgs, name_snake: &str) -> Result<()> {
    let builder_path = "apps/desktop/src/lib/services/operationArgsBuilder.ts";
    if !Path::new(builder_path).exists() {
        return Ok(());
    }
    let content = fs::read_to_string(builder_path)?;
    if content.contains(&format!("case '{}':", name_snake)) {
        return Ok(());
    }

    let ext = args.output_format.to_lowercase();
    let arm = format!(
        r#"        case '{name_snake}':
            args = {{
                input: docPath,
                output: customOutputPath || getOutputPath(`${{docName.replace(/\.[^.]+$/i, '')}}_output.{ext}`)
            }};
            break;
"#,
        name_snake = name_snake,
        ext = ext
    );

    // Insert before "default:"
    if let Some(pos) = content.find("        default:") {
        let mut new_content = content[..pos].to_string();
        new_content.push_str(&arm);
        new_content.push_str(&content[pos..]);
        write_or_print(builder_path, &new_content, args.dry_run)?;
    }
    Ok(())
}
```
Call `append_desktop_args_builder(args, &name_snake)?;` in `generate_all()`.

### Step 3: Universal Audit & Report Visualizer (`AuditReportViewer.svelte`)

Create `apps/desktop/src/lib/components/ui/AuditReportViewer.svelte`:
```svelte
<script lang="ts">
  interface Props {
    title: string;
    report: Record<string, any>;
    outputPath?: string;
  }
  let { title, report, outputPath }: Props = $props();
  let copied = $state(false);

  async function copyJson() {
    await navigator.clipboard.writeText(JSON.stringify(report, null, 2));
    copied = true;
    setTimeout(() => copied = false, 2000);
  }
</script>

<div class="audit-viewer" style="background: var(--bg-surface); border: 1px solid var(--border-color); border-radius: 8px; padding: 12px; margin-top: 12px;">
  <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 8px;">
    <h4 style="margin: 0; font-size: 13px; font-weight: 600;">📊 {title} Findings</h4>
    <button onclick={copyJson} style="font-size: 11px; padding: 4px 8px; border-radius: 4px; border: 1px solid var(--border-color); background: var(--bg-subtle);">
      {copied ? '✓ Copied' : 'Copy JSON'}
    </button>
  </div>
  <pre style="font-size: 11px; background: var(--bg-canvas); padding: 8px; border-radius: 6px; overflow-x: auto; max-height: 200px;">{JSON.stringify(report, null, 2)}</pre>
</div>
```

Integrate into `apps/desktop/src/lib/components/layout/OperationsPanel.svelte` inside the `lastExecution` result box when output format is `.json` or `.JSON`.
