### Step 2: Implementation Details (`paperpilot-pdf/src/operations/pdf_stamp.rs`)

```rust
use lopdf::{Document, Object, Dictionary, Stream};
use paperpilot_core::error::PdfError;
use paperpilot_core::traits::{PdfDocument, PdfOperation};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PdfStampOptions {
    /// Absolute or relative path to the image stamp (PNG/JPEG)
    #[serde(default)]
    pub image_path: Option<String>,
    /// Inline base64 image data (if path is not used)
    #[serde(default)]
    pub image_data: Option<String>,
    /// X coordinate from bottom-left origin in points (default: 50.0)
    #[serde(default = "default_x")]
    pub x: f32,
    /// Y coordinate from bottom-left origin in points (default: 50.0)
    #[serde(default = "default_y")]
    pub y: f32,
    /// Rendered stamp width in points (default: 150.0)
    #[serde(default = "default_width")]
    pub width: f32,
    /// Rendered stamp height in points (default: 50.0)
    #[serde(default = "default_height")]
    pub height: f32,
    /// Opacity between 0.0 (transparent) and 1.0 (opaque) (default: 1.0)
    #[serde(default = "default_opacity")]
    pub opacity: f32,
    /// Target pages: "all", "1", "1,3,5", "last" (default: "all")
    #[serde(default = "default_pages")]
    pub pages: String,
}

fn default_x() -> f32 { 50.0 }
fn default_y() -> f32 { 50.0 }
fn default_width() -> f32 { 150.0 }
fn default_height() -> f32 { 50.0 }
fn default_opacity() -> f32 { 1.0 }
fn default_pages() -> String { "all".to_string() }

impl Default for PdfStampOptions {
    fn default() -> Self {
        Self {
            image_path: None,
            image_data: None,
            x: default_x(),
            y: default_y(),
            width: default_width(),
            height: default_height(),
            opacity: default_opacity(),
            pages: default_pages(),
        }
    }
}
```

#### Stamping Algorithm
1. Parse image bytes from either `options.image_path` (using `std::fs::read`), `options.image_data` (base64 decode), or fallback to a default stamp raster (or use `tests/e2e_fixtures/real/img1.png` if path is unspecified).
2. Decode image with `image::load_from_memory(&img_bytes)` to obtain dimensions (`w`, `h`) and raw RGB8 or RGBA8 bytes.
3. Add image as an XObject `/Image` in the lopdf `Document`:
   ```rust
   let mut img_dict = Dictionary::new();
   img_dict.set("Type", Object::Name(b"XObject".to_vec()));
   img_dict.set("Subtype", Object::Name(b"Image".to_vec()));
   img_dict.set("Width", Object::Integer(w as i64));
   img_dict.set("Height", Object::Integer(h as i64));
   img_dict.set("ColorSpace", Object::Name(b"DeviceRGB".to_vec()));
   img_dict.set("BitsPerComponent", Object::Integer(8));
   ```
4. Register the image object ID in the page `/Resources` -> `/XObject` dictionary (e.g. `ImStamp`).
5. For target pages (filtered by `options.pages`), append the PDF drawing operators to `/Contents`:
   ```
   q
   {width} 0 0 {height} {x} {y} cm
   /ImStamp Do
   Q
   ```
6. Return `doc.save_to(&mut buf)` bytes.
