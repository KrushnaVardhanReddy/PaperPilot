```rust
use paperpilot_core::error::PdfError;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TableExtractOptions {
    #[serde(default = "default_pages")]

}

fn default_pages() -> String { "all".to_string() }
fn default_format() -> String { "json".to_string() }

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ExtractedTable {
    pub page: u32,

    pub cols: usize,

    let csv_output = if options.format.to_lowercase() == "csv" {

    } else {

        total_tables: tables.len(),

fn serialize_tables_to_csv(tables: &[ExtractedTable]) -> String {
    let mut out = String::new();
    for (i, table) in tables.iter().enumerate() {
        if i > 0 {
            out.push_str("\n\n");
        }
        if !table.headers.is_empty() {
            let escaped_headers: Vec<String> = table.headers.iter().map(|h| escape_csv_cell(h)).collect();
            out.push_str(&escaped_headers.join(","));
            out.push('\n');
        }
        for row in &table.data {
            let escaped_row: Vec<String> = row.iter().map(|cell| escape_csv_cell(cell)).collect();
            out.push_str(&escaped_row.join(","));
            out.push('\n');
        }
    }
    out
}

fn escape_csv_cell(val: &str) -> String {
    if val.contains(',') || val.contains('"') || val.contains('\n') {
        format!("\"{}\"", val.replace('"', "\"\""))
    } else {
        val.to_string()
    }
}
```

## 2. THE 5 THIN INTERFACE WRAPPERS

```rust

    #[arg(short, long)]

    format: String,
    #[arg(long, default_value = "all")]

}
```
Command handler reads input bytes $\to$ calls `extract_tables_bytes` $\to$ prints or writes JSON / CSV.

### B. MCP Server (`paperpilot-mcp/src/server.rs`)
Register tool `pdf_extract_tables`:
- Input schema: `input` (string, required), `output` (string, optional), `format` (string: "json" | "csv", optional), `pages` (string, optional).
- Call tool arm executes `extract_tables_bytes` and returns structured JSON text.

### C. REST Gateway (`paperpilot-gateway`)
- Route: `POST /api/v1/pdf/extract-tables`
- Accepts multipart file upload or JSON `{ input, format, pages }`.
- Returns `TableExtractResult` JSON or `text/csv` stream.

### D. Browser WASM Engine (`paperpilot-wasm`)
- In `operations.rs` & `lib.rs`:
```rust
#[wasm_bindgen]
pub fn extract_tables(input_bytes: &[u8], options_json: &str) -> Result<String, JsValue> {
    let options: TableExtractOptions = serde_json::from_str(options_json).unwrap_or_default();
    let result = operations::extract_tables(input_bytes, &options)
        .map_err(|e| JsValue::from_str(&e.to_string()))?;
    serde_json::to_string(&result).map_err(|e| JsValue::from_str(&e.to_string()))
}
```
