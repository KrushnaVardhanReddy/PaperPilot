## STEP 2: Update `apps/desktop/tests/e2e_ui.spec.ts`

Add test suites covering each Phase 4.F feature with proper async waiting and realistic inputs:

### 2.1 — Multi-Document Tabs
```typescript
test('can open multiple documents into separate tabs and switch between them', async ({ page }) => {
  await page.goto('/');
  const fileInput = page.locator('input[type="file"]').first();
  await fileInput.setInputFiles([
    path.resolve('../../tests/e2e_fixtures/single_page.pdf'),
    path.resolve('../../tests/e2e_fixtures/multi_page.pdf')
  ]);

  await expect(page.locator('#tab-doc-0')).toBeVisible({ timeout: 5000 });
  await expect(page.locator('#tab-doc-1')).toBeVisible({ timeout: 5000 });

  // Switch to tab 1
  await page.click('#tab-doc-1');
  await expect(page.locator('#tab-doc-1')).toHaveClass(/active/);
});
```

### 2.2 — Canvas Search Bar (`Ctrl+F`)
```typescript
test('canvas search bar opens with shortcut and responds to search', async ({ page }) => {
  await page.goto('/');
  const fileInput = page.locator('input[type="file"]').first();
  await fileInput.setInputFiles(path.resolve('../../tests/e2e_fixtures/multi_page.pdf'));

  // Trigger search shortcut
  await page.keyboard.press('ControlOrMeta+f');
  const searchBar = page.locator('#pdf-search-bar');
  await expect(searchBar).toBeVisible({ timeout: 5000 });

  // Search for text
  await page.fill('#pdf-search-input', 'Paper');
  await page.keyboard.press('Enter');
  await expect(page.locator('#pdf-search-count')).toBeVisible();
});
```

### 2.3 — Categorized Tools Dock & Navigation
```typescript
test('tools dock filters operations and navigates to parameter inspector', async ({ page }) => {
  await page.goto('/');
  const searchInput = page.locator('#tool-search-input');
  await expect(searchInput).toBeVisible();

  // Search for watermark
  await searchInput.fill('watermark');
  const watermarkTool = page.locator('#tool-watermark');
  await expect(watermarkTool).toBeVisible();

  // Navigate to inspector and back
  await watermarkTool.click();
  await expect(page.locator('#btn-back-tools')).toBeVisible();
  await page.click('#btn-back-tools');
  await expect(searchInput).toBeVisible();
});
```

### 2.4 — Visual Split-Screen Diff Comparison
```typescript
test('visual diff comparison opens and switches modes', async ({ page }) => {
  await page.goto('/');
  const fileInput = page.locator('input[type="file"]').first();
  await fileInput.setInputFiles([
    path.resolve('../../tests/e2e_fixtures/single_page.pdf'),
    path.resolve('../../tests/e2e_fixtures/multi_page.pdf')
  ]);

  await page.keyboard.press('ControlOrMeta+d');
  const diffContainer = page.locator('#visual-diff-container');
  if (await diffContainer.isVisible()) {
    await expect(page.locator('#diff-split-handle')).toBeVisible();
    await page.click('#diff-close-btn');
    await expect(diffContainer).not.toBeVisible();
  }
});
```
