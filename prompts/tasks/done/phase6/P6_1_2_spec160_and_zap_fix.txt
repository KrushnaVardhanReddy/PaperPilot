The workflow launches it and curls `http://localhost:8080`:
```yaml

        run: |
          PORT=8080 ./target/release/paperpilot-gateway &
          echo "Waiting for PaperPilot Gateway to become healthy..."
          for i in {1..30}; do
            if curl -s http://localhost:8080/health | grep -q "ok"; then
              echo "Gateway is healthy and listening on port 8080"
              break
            fi
            sleep 1
          done
```
