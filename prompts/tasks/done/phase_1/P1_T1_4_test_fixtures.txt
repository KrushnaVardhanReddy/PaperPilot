fn get_fixture_path(name: &str) -> PathBuf {

### Step 3: Generate fixtures as part of the build / use `include_bytes!` alternative

Since lopdf-generated PDFs need to be written to disk, the cleanest approach is:

1. The `make-fixtures` binary, when run, writes all fixtures to `tests/fixtures/` and `paperpilot-pdf/tests/fixtures/`.
2. Add a Cargo build script `build.rs` in `paperpilot-pdf/` that detects if fixtures are missing and calls `make-fixtures` automatically:

```rust
// paperpilot-pdf/build.rs
use std::path::Path;
use std::process::Command;

fn main() {
    let fixture_path = Path::new(env!("CARGO_MANIFEST_DIR"))
        .parent()
        .unwrap()
        .join("tests/fixtures/simple.pdf");

    if !fixture_path.exists() {
        // Attempt to run make-fixtures
        let status = Command::new("cargo")
            .args(["run", "--bin", "make-fixtures", "--manifest-path",
                   &format!("{}/tools/make-fixtures/Cargo.toml",
                             env!("CARGO_MANIFEST_DIR").replace("/paperpilot-pdf", ""))])
            .status();
        if let Err(e) = status {
            println!("cargo:warning=Could not auto-generate fixtures: {e}");
        }
    }

    println!("cargo:rerun-if-changed=tests/fixtures/");
}
```

**Alternative (preferred for simplicity):** Instead of a build script, just add a `[dev-dependencies]` workspace-level script and document in `README-TEST.md` that you must run `cargo run --bin make-fixtures` before running tests for the first time.

### Step 4: Generate the fixture PDFs correctly using lopdf

Here is a reference for creating a valid minimal PDF with lopdf:

```rust
use lopdf::{Document, Object, Stream, Dictionary, content::{Content, Operation}};

fn create_simple_pdf(text: &str) -> Document {
    let mut doc = Document::with_version("1.5");

    let pages_id = doc.new_object_id();
    let font_id = doc.add_object(dictionary! {
        "Type" => "Font",
        "Subtype" => "Type1",
        "BaseFont" => "Helvetica",
    });

    let content = Content {
        operations: vec![
            Operation::new("BT", vec![]),
            Operation::new("Tf", vec!["F1".into(), 14.into()]),
            Operation::new("Td", vec![50.into(), 700.into()]),
            Operation::new("Tj", vec![Object::string_literal(text)]),
            Operation::new("ET", vec![]),
        ],
    };
    let content_bytes = content.encode().unwrap();
    let content_id = doc.add_object(Stream::new(dictionary!{}, content_bytes));

    let page_id = doc.add_object(dictionary! {
        "Type" => "Page",
        "Parent" => pages_id,
        "MediaBox" => vec![0.into(), 0.into(), 612.into(), 792.into()],
        "Contents" => content_id,
        "Resources" => dictionary! {
            "Font" => dictionary! {
                "F1" => font_id,
            }
        }
    });

    doc.objects.insert(pages_id, Object::Dictionary(dictionary! {
        "Type" => "Pages",
        "Kids" => vec![page_id.into()],
        "Count" => 1,
    }));

    let catalog_id = doc.add_object(dictionary! {
        "Type" => "Catalog",
        "Pages" => pages_id,
    });
    doc.trailer.set("Root", catalog_id);
    doc
}
```

Use this pattern to create all fixtures. For `multi_page.pdf` (3 pages), extend `Pages.Kids` to have 3 page object IDs and set `Count` to 3.

For `malformed.pdf` do **NOT** use lopdf. Instead:
```rust
std::fs::write(output_path, b"%PDF-1.4\n1 0 obj\n<<CORRUPT GARBAGE HERE>>\nxref\n%%EOF")?;
```
