STEP 1: Run Button Loading State (OperationsPanel.svelte)
- Add state variables near existing $state() in <script>:
  let isRunning = $state(false);
  let elapsedSeconds = $state(0);
  let elapsedDisplay = $derived(
    `${String(Math.floor(elapsedSeconds / 60)).padStart(2, '0')}:${String(elapsedSeconds % 60).padStart(2, '0')}`
  );

- In `handleRunOperation`, wrap the execution in try/finally:
  async function handleRunOperation() {
    if (isRunning) return;
    isRunning = true;
    elapsedSeconds = 0;
    const timer = setInterval(() => { elapsedSeconds++; }, 1000);
    try {
      // ... existing body unchanged ...
    } finally {
      isRunning = false;
      clearInterval(timer);
    }
  }

- In the template, update the run button:
  <div class="action-area">
    <button
      class="run-btn"
      class:run-btn--loading={isRunning}
      id="btn-run-operation"
      disabled={isRunning || !activeTool || (!['md_to_pdf','html_to_pdf','img_to_pdf','json_to_pdf'].includes(activeTool.id) && appState.documents.length === 0)}
      onclick={handleRunOperation}
    >
      {#if isRunning}
        <span class="spinner" aria-hidden="true"></span>
        Running...
      {:else}
        Run {activeTool.title}
      {/if}
    </button>
    {#if isRunning}
      <span class="elapsed-time" aria-live="polite">{elapsedDisplay}</span>
    {/if}
  </div>

- Add CSS for .spinner, .run-btn--loading, .elapsed-time.
