```typescript
  if_scanned?: boolean;

  value?: number;

export type WorkflowCondition =
  | { all: WorkflowCondition[] }
```

3. Update `setStepCondition` in `PipelineState`:
```typescript
setStepCondition(stepId: string, condition?: WorkflowCondition, onNoMatch?: OnNoMatch) {
  const step = this.steps.find(s => s.id === stepId);
  if (!step) return;
  step.condition = condition;
  if (onNoMatch) {
    step.on_no_match = onNoMatch;
  }
}
```

#### Step 2: Implement Multi-Rule UI in `apps/desktop/src/lib/components/pipeline/ConditionInspector.svelte`
1. Add an editable UI rule data structure:
```typescript
export type PredicateType = 'scanned' | 'page_count' | 'text_contains' | 'has_form_fields' | 'is_signed';

export interface UiRule {
  id: string;
  type: PredicateType;
  comparator?: 'Greater' | 'GreaterOrEqual' | 'Equal' | 'Less' | 'LessOrEqual';
  numValue?: number;
  textValue?: string;
  boolValue?: boolean;
}
```

2. Normalization & Serialization helpers:
- When a step is selected, parse `step.condition`:
  - If `{ all: items }`: combinator = `'all'`, rules = parsed items.
  - If `{ any: items }`: combinator = `'any'`, rules = parsed items.
  - If flat leaf: combinator = `'all'`, rules = single parsed item.
- When saving:
  - If rules list is empty: condition is `undefined`.
  - If 1 rule and combinator is `'all'`: can serialize as flat leaf `{ if_scanned: ... }` for maximum backward compatibility.
  - If multiple rules with combinator `'all'`: serialize as `{ all: [leaf1, leaf2, ...] }`.
  - If rules with combinator `'any'`: serialize as `{ any: [leaf1, leaf2, ...] }`.

3. Template Structure:
- **Combinator Header**:

  ```svelte
      <option value="all">ALL rules match (AND)</option>
      <option value="any">ANY rule matches (OR)</option>
    </select>
  </div>
  ```
- **Rules List (`{#each rules as rule, index (rule.id)}`)**:
  - Predicate selector `<select>`:
